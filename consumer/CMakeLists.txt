cmake_minimum_required(VERSION 3.21)
project(DesktopInstalledConsumer LANGUAGES C CXX)
find_package(WeappContainerDesktop CONFIG REQUIRED)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  add_executable(desktop_consumer macos.mm c_smoke.c)
  set_source_files_properties(macos.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
elseif(WIN32)
  add_executable(desktop_consumer windows.cpp c_smoke.c)
  target_compile_definitions(desktop_consumer PRIVATE _WIN32_WINNT=0x0A00 NOMINMAX)
else()
  add_executable(desktop_consumer linux.cpp c_smoke.c)
endif()
target_link_libraries(desktop_consumer PRIVATE WeappContainer::Desktop)

option(WEAPP_CONSUMER_SHOWCASE "Run restricted business-pack acceptance" OFF)
if(WEAPP_CONSUMER_SHOWCASE)
  target_compile_definitions(desktop_consumer PRIVATE WEAPP_CONSUMER_SHOWCASE=1)
endif()
