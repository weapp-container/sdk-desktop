cmake_minimum_required(VERSION 3.21)
project(WeappContainerDesktop VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(WEAPP_CORE_DIR "" CACHE PATH "Directory containing the prebuilt native SDK archive and header")
set(WEAPP_DESKTOP_USES_CURL OFF)
set(WEAPP_DESKTOP_CURL_MIN_VERSION 7.85)
if(NOT WEAPP_CORE_DIR)
  message(FATAL_ERROR "Set WEAPP_CORE_DIR to the verified private runtime distribution; consumers do not build Rust")
endif()
if(WIN32)
  set(WEAPP_CORE_FILE weapp_container_sdk.lib)
else()
  set(WEAPP_CORE_FILE libweapp_container_sdk.a)
endif()
if(NOT EXISTS "${WEAPP_CORE_DIR}/${WEAPP_CORE_FILE}" OR NOT EXISTS "${WEAPP_CORE_DIR}/weapp_container_sdk.h")
  message(FATAL_ERROR "WEAPP_CORE_DIR lacks the native static library or public C header")
endif()
add_library(WeappContainer::Core STATIC IMPORTED GLOBAL)
set_target_properties(WeappContainer::Core PROPERTIES IMPORTED_LOCATION "${WEAPP_CORE_DIR}/${WEAPP_CORE_FILE}")
add_library(weapp_container_desktop STATIC src/desktop.cpp src/c_api.cpp src/native_preview.cpp)
add_library(WeappContainer::Desktop ALIAS weapp_container_desktop)
set_target_properties(weapp_container_desktop PROPERTIES EXPORT_NAME Desktop POSITION_INDEPENDENT_CODE ON)
target_compile_features(weapp_container_desktop PUBLIC cxx_std_17)
target_include_directories(weapp_container_desktop PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_include_directories(weapp_container_desktop PRIVATE third_party "${WEAPP_CORE_DIR}")
target_link_libraries(weapp_container_desktop PUBLIC WeappContainer::Core)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  target_sources(weapp_container_desktop PRIVATE src/macos.mm src/device_apis_macos.mm)
  set_source_files_properties(src/macos.mm src/device_apis_macos.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_link_libraries(weapp_container_desktop PUBLIC
    "-framework AppKit" "-framework Metal" "-framework QuartzCore" "-framework CoreGraphics"
    "-framework CoreText" "-framework CoreFoundation" "-framework Foundation" "-framework IOSurface"
    "-framework Security" "-framework SystemConfiguration" "-framework IOKit"
    "-framework AVFoundation" "-framework CoreMedia" "-framework CoreVideo"
    "-framework CoreLocation" "-framework CoreWLAN" "-framework CoreImage"
    "-framework CoreBluetooth" "-framework Contacts" "-framework EventKit" "-framework LocalAuthentication" c++ iconv resolv)
elseif(WIN32)
  # Current C++/WinRT projections use standard coroutines. Apply the same
  # language mode to feature probes and the native implementation so an SDK
  # does not disappear merely because its headers were checked in C++17 mode.
  set(CMAKE_CXX_STANDARD 20)
  target_compile_features(weapp_container_desktop PRIVATE cxx_std_20)
  target_sources(weapp_container_desktop PRIVATE src/windows.cpp src/device_apis_windows.cpp)
  target_compile_definitions(weapp_container_desktop PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  include(CheckIncludeFileCXX)
  check_include_file_cxx("winrt/Windows.Devices.Geolocation.h" WEAPP_HAS_WINRT_LOCATION)
  check_include_file_cxx("winrt/Windows.Devices.Bluetooth.h" WEAPP_HAS_WINRT_BLUETOOTH)
  if(WEAPP_HAS_WINRT_LOCATION)
    target_compile_definitions(weapp_container_desktop PRIVATE WEAPP_DEVICE_WINRT_LOCATION=1)
  endif()
  if(WEAPP_HAS_WINRT_BLUETOOTH)
    target_compile_definitions(weapp_container_desktop PRIVATE WEAPP_DEVICE_WINRT_BLUETOOTH=1)
  endif()
  if(WEAPP_HAS_WINRT_LOCATION OR WEAPP_HAS_WINRT_BLUETOOTH)
    target_link_libraries(weapp_container_desktop PUBLIC windowsapp runtimeobject)
  endif()
  target_link_libraries(weapp_container_desktop PUBLIC user32 gdi32 msimg32 dwrite usp10 windowscodecs opengl32
    ws2_32 bcrypt ntdll userenv advapi32 crypt32 secur32 shell32 ole32 oleaut32 shlwapi winhttp comctl32 wlanapi iphlpapi)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0>=3.24 gio-2.0>=2.66 fontconfig epoxy)
  target_sources(weapp_container_desktop PRIVATE src/linux.cpp src/device_apis_linux.cpp)
  target_link_libraries(weapp_container_desktop PUBLIC PkgConfig::GTK3 dl pthread m)
  find_package(CURL 7.85 QUIET)
  if(CURL_FOUND)
    set(WEAPP_DESKTOP_USES_CURL ON)
    target_compile_definitions(weapp_container_desktop PRIVATE WEAPP_DEVICE_CURL=1)
    target_link_libraries(weapp_container_desktop PUBLIC CURL::libcurl)
    # Earlier WebSocket APIs have incompatible short-write accounting. The
    # wrapper relies on the frame/partial-write behavior provided in 8.16+.
    if(CURL_VERSION_STRING VERSION_GREATER_EQUAL "8.16")
      include(CheckCXXSourceCompiles)
      set(CMAKE_REQUIRED_LIBRARIES CURL::libcurl)
      check_cxx_source_compiles("#include <curl/curl.h>\n#include <curl/websockets.h>\nint main(){auto volatile send=&curl_ws_send;auto volatile recv=&curl_ws_recv;return send==nullptr||recv==nullptr;}" WEAPP_CURL_HAS_WEBSOCKET_API)
      unset(CMAKE_REQUIRED_LIBRARIES)
      if(WEAPP_CURL_HAS_WEBSOCKET_API)
        set(WEAPP_DESKTOP_CURL_MIN_VERSION 8.16)
        target_compile_definitions(weapp_container_desktop PRIVATE WEAPP_DEVICE_CURL_WEBSOCKET=1)
      endif()
    endif()
  endif()
endif()
option(WEAPP_DESKTOP_BUILD_TESTS "Build owner-thread native service contract regressions" OFF)
if(WEAPP_DESKTOP_BUILD_TESTS)
  enable_testing()
  add_executable(weapp_device_permissions_test test/device_permissions.cpp)
  target_include_directories(weapp_device_permissions_test PRIVATE src third_party "${WEAPP_CORE_DIR}")
  target_link_libraries(weapp_device_permissions_test PRIVATE WeappContainer::Desktop)
  add_test(NAME desktop_device_permissions COMMAND weapp_device_permissions_test)
  add_executable(weapp_device_publication_test test/device_publication.cpp)
  target_include_directories(weapp_device_publication_test PRIVATE src third_party "${WEAPP_CORE_DIR}")
  target_link_libraries(weapp_device_publication_test PRIVATE WeappContainer::Desktop)
  add_test(NAME desktop_device_publication COMMAND weapp_device_publication_test)
  add_executable(weapp_device_socket_test test/device_socket.cpp)
  target_include_directories(weapp_device_socket_test PRIVATE src third_party "${WEAPP_CORE_DIR}")
  target_link_libraries(weapp_device_socket_test PRIVATE WeappContainer::Desktop)
  add_test(NAME desktop_device_socket COMMAND weapp_device_socket_test)
  add_executable(weapp_device_bluetooth_test test/device_bluetooth.cpp)
  target_include_directories(weapp_device_bluetooth_test PRIVATE src third_party "${WEAPP_CORE_DIR}")
  target_link_libraries(weapp_device_bluetooth_test PRIVATE WeappContainer::Desktop)
  add_test(NAME desktop_device_bluetooth COMMAND weapp_device_bluetooth_test)
  add_executable(weapp_device_location_test test/device_location.cpp)
  target_include_directories(weapp_device_location_test PRIVATE src third_party "${WEAPP_CORE_DIR}")
  target_link_libraries(weapp_device_location_test PRIVATE WeappContainer::Desktop)
  add_test(NAME desktop_device_location COMMAND weapp_device_location_test)
  add_executable(weapp_device_socket_protocol_test test/device_socket_protocol.cpp)
  target_include_directories(weapp_device_socket_protocol_test PRIVATE src include third_party "${WEAPP_CORE_DIR}")
  find_package(Threads REQUIRED)
  target_link_libraries(weapp_device_socket_protocol_test PRIVATE Threads::Threads)
  add_test(NAME desktop_device_socket_protocol COMMAND weapp_device_socket_protocol_test)
  if(WIN32)
    foreach(native_test weapp_device_bluetooth_test weapp_device_permissions_test weapp_device_publication_test weapp_device_socket_test weapp_device_socket_protocol_test weapp_device_location_test)
      target_compile_definitions(${native_test} PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    endforeach()
  endif()
  if(UNIX)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
      pkg_check_modules(GIO QUIET IMPORTED_TARGET gio-2.0>=2.66)
      if(GIO_FOUND)
        add_library(weapp_bluez_compile OBJECT test/bluez_compile.cpp)
        target_include_directories(weapp_bluez_compile PRIVATE src include third_party "${WEAPP_CORE_DIR}")
        target_link_libraries(weapp_bluez_compile PRIVATE PkgConfig::GIO)
        add_library(weapp_geoclue_compile OBJECT test/geoclue_compile.cpp)
        target_include_directories(weapp_geoclue_compile PRIVATE src include third_party "${WEAPP_CORE_DIR}")
        target_link_libraries(weapp_geoclue_compile PRIVATE PkgConfig::GIO)
      endif()
    endif()
    find_package(CURL 8.16 QUIET)
    if(CURL_FOUND)
      add_library(weapp_curl_socket_compile OBJECT test/curl_socket_compile.cpp)
      target_include_directories(weapp_curl_socket_compile PRIVATE src include third_party "${WEAPP_CORE_DIR}")
      target_link_libraries(weapp_curl_socket_compile PRIVATE CURL::libcurl)
      add_executable(weapp_curl_socket_headers_test test/curl_socket_headers.cpp)
      target_include_directories(weapp_curl_socket_headers_test PRIVATE src include third_party "${WEAPP_CORE_DIR}")
      target_link_libraries(weapp_curl_socket_headers_test PRIVATE CURL::libcurl Threads::Threads)
      add_test(NAME desktop_curl_socket_headers COMMAND weapp_curl_socket_headers_test)
    endif()
  endif()
endif()
install(TARGETS weapp_container_desktop EXPORT WeappContainerDesktopTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES "${WEAPP_CORE_DIR}/${WEAPP_CORE_FILE}" DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES "${WEAPP_CORE_DIR}/weapp_container_sdk.h" DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION ${CMAKE_INSTALL_DATADIR}/weapp-container/licenses RENAME nlohmann-json-LICENSE.MIT)
install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/weapp-container)
install(DIRECTORY consumer/ DESTINATION ${CMAKE_INSTALL_DATADIR}/weapp-container/consumer)
configure_package_config_file(cmake/WeappContainerDesktopConfig.cmake.in
  "${CMAKE_CURRENT_BINARY_DIR}/WeappContainerDesktopConfig.cmake"
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/WeappContainerDesktop)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/WeappContainerDesktopConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT WeappContainerDesktopTargets NAMESPACE WeappContainer:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/WeappContainerDesktop)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/WeappContainerDesktopConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/WeappContainerDesktopConfigVersion.cmake"
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/WeappContainerDesktop)
