cmake_minimum_required(VERSION 3.21)
project(WeappContainerDesktop VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(WEAPP_CORE_DIR "" CACHE PATH "Directory containing the prebuilt native SDK archive and header")
set(WEAPP_DESKTOP_USES_CURL OFF)
set(WEAPP_DESKTOP_CURL_MIN_VERSION 7.85)
if(NOT WEAPP_CORE_DIR)
  message(FATAL_ERROR "Set WEAPP_CORE_DIR to the verified private runtime distribution; consumers do not build Rust")
endif()
if(WIN32)
  set(WEAPP_CORE_FILE weapp_container_sdk.dll.lib)
  set(WEAPP_CORE_RUNTIME_FILE weapp_container_sdk.dll)
else()
  set(WEAPP_CORE_FILE libweapp_container_sdk.a)
endif()
if(NOT EXISTS "${WEAPP_CORE_DIR}/${WEAPP_CORE_FILE}" OR NOT EXISTS "${WEAPP_CORE_DIR}/weapp_container_sdk.h")
  message(FATAL_ERROR "WEAPP_CORE_DIR lacks the native link library or public C header")
endif()
if(WIN32)
  if(NOT EXISTS "${WEAPP_CORE_DIR}/${WEAPP_CORE_RUNTIME_FILE}")
    message(FATAL_ERROR "WEAPP_CORE_DIR lacks the verified native runtime DLL")
  endif()
  add_library(WeappContainer::Core SHARED IMPORTED GLOBAL)
  set_target_properties(WeappContainer::Core PROPERTIES IMPORTED_IMPLIB "${WEAPP_CORE_DIR}/${WEAPP_CORE_FILE}"
    IMPORTED_LOCATION "${WEAPP_CORE_DIR}/${WEAPP_CORE_RUNTIME_FILE}" INTERFACE_COMPILE_DEFINITIONS WEAPP_CONTAINER_SDK_SHARED)
else()
  add_library(WeappContainer::Core STATIC IMPORTED GLOBAL)
  set_target_properties(WeappContainer::Core PROPERTIES IMPORTED_LOCATION "${WEAPP_CORE_DIR}/${WEAPP_CORE_FILE}")
endif()
add_library(weapp_container_desktop STATIC src/desktop.cpp src/value.cpp src/c_api.cpp src/native_preview.cpp
  src/host_cancellation.cpp src/host_services.cpp src/host_services_adapter.cpp)
add_library(WeappContainer::Desktop ALIAS weapp_container_desktop)
set_target_properties(weapp_container_desktop PROPERTIES EXPORT_NAME Desktop POSITION_INDEPENDENT_CODE ON)
target_compile_features(weapp_container_desktop PUBLIC cxx_std_17)
target_include_directories(weapp_container_desktop PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_include_directories(weapp_container_desktop PRIVATE third_party "${WEAPP_CORE_DIR}")
target_link_libraries(weapp_container_desktop PUBLIC WeappContainer::Core)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  target_sources(weapp_container_desktop PRIVATE src/macos.mm src/device_apis_macos.mm src/host_services_macos.mm)
  set_source_files_properties(src/macos.mm src/device_apis_macos.mm src/host_services_macos.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_link_libraries(weapp_container_desktop PUBLIC
    "-framework AppKit" "-framework Metal" "-framework QuartzCore" "-framework CoreGraphics"
    "-framework CoreText" "-framework CoreFoundation" "-framework Foundation" "-framework IOSurface"
    "-framework Security" "-framework SystemConfiguration" "-framework IOKit"
    "-framework AVFoundation" "-framework CoreMedia" "-framework CoreVideo"
    "-framework CoreLocation" "-framework CoreWLAN" "-framework CoreImage"
    "-framework CoreBluetooth" "-framework Contacts" "-framework EventKit" "-framework LocalAuthentication" c++ iconv resolv)
elseif(WIN32)
  # Current C++/WinRT projections use standard coroutines. Apply the same
  # language mode to feature probes and the native implementation so an SDK
  # does not disappear merely because its headers were checked in C++17 mode.
  set(CMAKE_CXX_STANDARD 20)
  target_compile_features(weapp_container_desktop PRIVATE cxx_std_20)
  if(MSVC)
    target_compile_options(weapp_container_desktop PRIVATE /utf-8 /experimental:deterministic)
    # MSVC emits object names even in Release, and C++/WinRT expands
    # __FILE__ in failure diagnostics. Remap both before distributing .lib.
    target_compile_options(weapp_container_desktop PRIVATE
      "/pathmap:${CMAKE_SOURCE_DIR}=weapp-container/sdk-desktop"
      "/pathmap:${CMAKE_BINARY_DIR}=weapp-desktop-build")
    if(DEFINED ENV{WindowsSdkDir})
      file(TO_CMAKE_PATH "$ENV{WindowsSdkDir}" WEAPP_WINDOWS_SDK_DIR)
      target_compile_options(weapp_container_desktop PRIVATE
        "/pathmap:${WEAPP_WINDOWS_SDK_DIR}=toolchains/windows-sdk/")
    endif()
  endif()
  target_sources(weapp_container_desktop PRIVATE src/windows.cpp src/device_apis_windows.cpp src/host_services_windows.cpp)
  target_compile_definitions(weapp_container_desktop PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  include(CheckIncludeFileCXX)
  check_include_file_cxx("winrt/Windows.Devices.Geolocation.h" WEAPP_HAS_WINRT_LOCATION)
  check_include_file_cxx("winrt/Windows.Devices.Bluetooth.h" WEAPP_HAS_WINRT_BLUETOOTH)
  check_include_file_cxx("winrt/Windows.Devices.Sensors.h" WEAPP_HAS_WINRT_SENSORS)
  check_include_file_cxx("winrt/Windows.Media.Playback.h" WEAPP_HAS_WINRT_AUDIO)
  if(WEAPP_HAS_WINRT_AUDIO)
    target_compile_definitions(weapp_container_desktop PRIVATE WEAPP_DEVICE_WINRT_AUDIO=1)
  endif()
  if(WEAPP_HAS_WINRT_LOCATION)
    target_compile_definitions(weapp_container_desktop PRIVATE WEAPP_DEVICE_WINRT_LOCATION=1)
  endif()
  if(WEAPP_HAS_WINRT_BLUETOOTH)
    target_compile_definitions(weapp_container_desktop PRIVATE WEAPP_DEVICE_WINRT_BLUETOOTH=1)
  endif()
  if(WEAPP_HAS_WINRT_SENSORS)
    target_compile_definitions(weapp_container_desktop PRIVATE WEAPP_DEVICE_WINRT_SENSORS=1)
  endif()
  if(WEAPP_HAS_WINRT_LOCATION OR WEAPP_HAS_WINRT_BLUETOOTH OR WEAPP_HAS_WINRT_SENSORS OR WEAPP_HAS_WINRT_AUDIO)
    target_link_libraries(weapp_container_desktop PUBLIC windowsapp runtimeobject)
  endif()
  target_link_libraries(weapp_container_desktop PUBLIC user32 gdi32 msimg32 dwrite usp10 windowscodecs opengl32
    ws2_32 bcrypt ntdll userenv advapi32 crypt32 secur32 shell32 ole32 oleaut32 shlwapi winhttp comctl32 wlanapi iphlpapi)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0>=3.24 gio-2.0>=2.66 fontconfig epoxy)
  target_sources(weapp_container_desktop PRIVATE src/linux.cpp src/device_apis_linux.cpp src/host_services_linux.cpp)
  target_link_libraries(weapp_container_desktop PUBLIC PkgConfig::GTK3 dl pthread m)
  find_package(CURL 7.85 QUIET)
  if(CURL_FOUND)
    set(WEAPP_DESKTOP_USES_CURL ON)
    target_compile_definitions(weapp_container_desktop PRIVATE WEAPP_DEVICE_CURL=1)
    target_link_libraries(weapp_container_desktop PUBLIC CURL::libcurl)
    # Earlier WebSocket APIs have incompatible short-write accounting. The
    # wrapper relies on the frame/partial-write behavior provided in 8.16+.
    if(CURL_VERSION_STRING VERSION_GREATER_EQUAL "8.16")
      include(CheckCXXSourceCompiles)
      set(CMAKE_REQUIRED_LIBRARIES CURL::libcurl)
      check_cxx_source_compiles("#include <curl/curl.h>\n#include <curl/websockets.h>\nint main(){auto volatile send=&curl_ws_send;auto volatile recv=&curl_ws_recv;return send==nullptr||recv==nullptr;}" WEAPP_CURL_HAS_WEBSOCKET_API)
      unset(CMAKE_REQUIRED_LIBRARIES)
      if(WEAPP_CURL_HAS_WEBSOCKET_API)
        set(WEAPP_DESKTOP_CURL_MIN_VERSION 8.16)
        target_compile_definitions(weapp_container_desktop PRIVATE WEAPP_DEVICE_CURL_WEBSOCKET=1)
      endif()
    endif()
  endif()
endif()
