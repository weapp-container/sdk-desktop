cmake_minimum_required(VERSION 3.21)
project(WeappDesktopNativeSources LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
get_filename_component(WEAPP_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
find_package(Threads REQUIRED)
add_executable(value_encoding ../value_encoding.cpp ../../src/value.cpp)
target_include_directories(value_encoding PRIVATE "${WEAPP_SOURCE_ROOT}/src" "${WEAPP_SOURCE_ROOT}/include" "${WEAPP_SOURCE_ROOT}/third_party")
add_test(NAME value_encoding COMMAND value_encoding)
add_executable(sensor_lifecycle ../device_sensor.cpp ../../src/value.cpp)
target_include_directories(sensor_lifecycle PRIVATE "${WEAPP_SOURCE_ROOT}/src" "${WEAPP_SOURCE_ROOT}/include" "${WEAPP_SOURCE_ROOT}/third_party")
target_link_libraries(sensor_lifecycle PRIVATE Threads::Threads)
add_test(NAME sensor_lifecycle COMMAND sensor_lifecycle)
add_executable(audio_lifecycle ../device_audio.cpp ../../src/value.cpp)
target_include_directories(audio_lifecycle PRIVATE "${WEAPP_SOURCE_ROOT}/src" "${WEAPP_SOURCE_ROOT}/include" "${WEAPP_SOURCE_ROOT}/third_party")
target_link_libraries(audio_lifecycle PRIVATE Threads::Threads)
add_test(NAME audio_lifecycle COMMAND audio_lifecycle)
if(WIN32)
  # This target uses the actual Windows SDK to compile every native wrapper
  # translation unit. An OBJECT target intentionally performs no core link:
  # private Windows runtime archives are not available in this source checkout.
  add_library(windows_native_sources OBJECT ../../src/windows.cpp ../../src/device_apis_windows.cpp ../../src/desktop.cpp ../../src/value.cpp ../../src/c_api.cpp ../../src/native_preview.cpp
    ../../src/host_services.cpp ../../src/host_services_adapter.cpp ../../src/host_services_windows.cpp ../host_services_reference.cpp winrt_sensors.cpp winrt_audio.cpp)
  target_include_directories(windows_native_sources PRIVATE "${WEAPP_SOURCE_ROOT}/src" "${WEAPP_SOURCE_ROOT}/include" "${WEAPP_SOURCE_ROOT}/third_party")
  target_compile_definitions(windows_native_sources PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00 WEAPP_DEVICE_WINRT_LOCATION=1 WEAPP_DEVICE_WINRT_BLUETOOTH=1 WEAPP_DEVICE_WINRT_SENSORS=1 WEAPP_DEVICE_WINRT_AUDIO=1)
  target_compile_definitions(sensor_lifecycle PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_compile_definitions(audio_lifecycle PRIVATE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(audio_lifecycle PRIVATE bcrypt ole32)
  # DeviceCommon's Win32 file helpers remain compiled in the finite test.
  target_link_libraries(sensor_lifecycle PRIVATE bcrypt ole32)
endif()
if(UNIX AND NOT APPLE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0>=3.24 gio-2.0>=2.66 fontconfig epoxy)
  find_package(CURL 7.85 REQUIRED)
  # Compile the real Linux wrapper against GTK/GIO/libcurl. This object target
  # makes no private runtime link or device/presentation acceptance claim.
  add_library(linux_native_sources OBJECT ../../src/linux.cpp ../../src/device_apis_linux.cpp ../../src/desktop.cpp ../../src/value.cpp ../../src/c_api.cpp ../../src/native_preview.cpp
    ../../src/host_services.cpp ../../src/host_services_adapter.cpp ../../src/host_services_linux.cpp ../host_services_reference.cpp)
  target_include_directories(linux_native_sources PRIVATE "${WEAPP_SOURCE_ROOT}/src" "${WEAPP_SOURCE_ROOT}/include" "${WEAPP_SOURCE_ROOT}/third_party")
  target_compile_definitions(linux_native_sources PRIVATE WEAPP_DEVICE_CURL=1)
  target_link_libraries(linux_native_sources PRIVATE PkgConfig::GTK3 CURL::libcurl)
  add_executable(linux_consent_thread ../linux_consent_thread.cpp ../../src/host_services_linux.cpp)
  target_include_directories(linux_consent_thread PRIVATE "${WEAPP_SOURCE_ROOT}/src" "${WEAPP_SOURCE_ROOT}/include" "${WEAPP_SOURCE_ROOT}/third_party")
  target_compile_definitions(linux_consent_thread PRIVATE WEAPP_DEVICE_CURL=0)
  target_link_libraries(linux_consent_thread PRIVATE PkgConfig::GTK3 Threads::Threads)
  add_test(NAME linux_consent_thread COMMAND linux_consent_thread)
endif()
